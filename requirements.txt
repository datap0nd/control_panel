-- Replace ROLE_NAME, SCHEMA_NAME, and TABLE_NAME before running.

SELECT
    current_database() AS connected_database,
    current_user AS executing_user,
    has_schema_privilege('ROLE_NAME', '"SCHEMA_NAME"', 'USAGE') AS schema_usage,
    has_table_privilege('ROLE_NAME', '"SCHEMA_NAME"."TABLE_NAME"', 'SELECT') AS table_select;

GRANT USAGE ON SCHEMA "SCHEMA_NAME" TO ROLE_NAME;
GRANT SELECT ON TABLE "SCHEMA_NAME"."TABLE_NAME" TO ROLE_NAME;

SELECT
    has_schema_privilege('ROLE_NAME', '"SCHEMA_NAME"', 'USAGE') AS schema_usage_after,
    has_table_privilege('ROLE_NAME', '"SCHEMA_NAME"."TABLE_NAME"', 'SELECT') AS table_select_after;
