SELECT
    current_user,
    session_user,
    current_database(),
    inet_server_addr(),
    inet_server_port();

SELECT
    pg_has_role(current_user, 'EXPECTED_ROLE', 'member') AS is_expected_role_or_member;

WITH target AS (
    SELECT
        to_regnamespace('TARGET_SCHEMA') AS schema_oid,
        to_regclass('"TARGET_SCHEMA"."TARGET_TABLE"') AS table_oid
)
SELECT
    current_user AS effective_user,
    schema_oid IS NOT NULL AS schema_visible,
    table_oid::text AS resolved_table,
    CASE
        WHEN schema_oid IS NULL THEN NULL
        ELSE has_schema_privilege(current_user, schema_oid, 'USAGE')
    END AS has_schema_usage,
    CASE
        WHEN table_oid IS NULL THEN NULL
        ELSE has_table_privilege(current_user, table_oid, 'SELECT')
    END AS has_table_select
FROM target;
